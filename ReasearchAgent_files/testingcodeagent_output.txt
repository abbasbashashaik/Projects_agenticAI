from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
from flask_marshmallow import Marshmallow
from flask_jwt_extended import JWTManager, jwt_required, get_jwt_identity
from marshmallow import Schema, fields
import os

app = Flask(__name__)
app.config["SQLALCHEMY_DATABASE_URI"] = os.environ.get("DATABASE_URL")
db = SQLAlchemy(app)
ma = Marshmallow(app)
jwt = JWTManager(app)


class TodoItem(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(100), nullable=False)
    completed = db.Column(db.Boolean, default=False)


class TodoItemSchema(ma.SQLAlchemyAutoSchema):
    class Meta:
        model = TodoItem


def create_todo_item(title):
    todo_item = TodoItem(title=title)
    db.session.add(todo_item)
    db.session.commit()
    return todo_item


@app.route("/todo", methods=["GET"])
def get_all_todo_items():
    todo_items = TodoItem.query.all()
    return TodoItemSchema(many=True).dump(todo_items)


@app.route("/todo", methods=["POST"])
def create_todo_item_endpoint():
    data = request.get_json()
    title = data.get("title")
    if not title:
        return {"error": "Missing title"}, 422
    todo_item = create_todo_item(title)
    return TodoItemSchema().dump(todo_item)


@app.route("/todo/<int:id>", methods=["GET"])
def get_todo_item_endpoint(id):
    todo_item = TodoItem.query.get(id)
    if not todo_item:
        return {"error": "Todo item not found"}, 404
    return TodoItemSchema().dump(todo_item)


@app.route("/todo/<int:id>", methods=["PUT"])
@jwt_required
def update_todo_item_endpoint(id):
    todo_item = TodoItem.query.get(id)
    if not todo_item:
        return {"error": "Todo item not found"}, 404
    data = request.get_json()
    title = data.get("title")
    if title:
        todo_item.title = title
        db.session.commit()
    return TodoItemSchema().dump(todo_item)


@app.route("/todo/<int:id>", methods=["DELETE"])
@jwt_required
def delete_todo_item_endpoint(id):
    todo_item = TodoItem.query.get(id)
    if not todo_item:
        return {"error": "Todo item not found"}, 404
    db.session.delete(todo_item)
    db.session.commit()

class TodoService:
    @staticmethod
    def create_todo_item(title):
        # Create a new todo item
        todo_item = TodoItem.query.filter_by(title=title).first()
        if todo_item:
            return {"error": "Todo item already exists"}
        db.session.add(TodoItem(title=title))
        db.session.commit()
        return todo_item

    @staticmethod
    def get_todo_item(id):
        # Get a todo item by ID
        todo_item = TodoItem.query.get(id)
        return todo_item

    @staticmethod
    def update_todo_item(title, id):
        # Update a todo item
        todo_item = TodoItem.query.get(id)
        if not todo_item:
            return {"error": "Todo item not found"}
        todo_item.title = title
        db.session.commit()
        return todo_item

    @staticmethod
    def delete_todo_item(id):
        # Delete a todo item
        todo_item = TodoItem.query.get(id)
        if not todo_item:
